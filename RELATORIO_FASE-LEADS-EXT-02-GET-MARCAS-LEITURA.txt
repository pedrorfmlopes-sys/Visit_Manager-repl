GET /api/crm/leads

Request:
  GET /api/crm/leads?empresaId=...&entidadeId=...

Query:
  1. SELECT leads... WHERE empresaId=...
  2. LEFT JOIN leads_marcas ON leads.id = leads_marcas.lead_id
  3. LEFT JOIN marcas ON leads_marcas.marca_id = marcas.id

Response (exemplo):
  {
    "leads": [
      {
        "id": "lead-1",
        "titulo": "Oportunidade A",
        "marca": "Ritmonio",  // campo texto existente (mantido)
        "marcas": [           // NOVO - array lido
          { "id": "marca-1", "nome": "Ritmonio" },
          { "id": "marca-2", "nome": "Bette" }
        ],
        "contactosAssociados": [...],
        ...
      },
      {
        "id": "lead-2",
        "titulo": "Oportunidade B",
        "marca": null,
        "marcas": [],  // lead antigo, sem marcas
        "contactosAssociados": [...],
        ...
      }
    ]
  }

GET /api/crm/leads/:id

Request:
  GET /api/crm/leads/lead-1

Query:
  1. SELECT leads... WHERE id=lead-1 AND empresaId=...

Response (exemplo):

      ...
    }
  }
