app.post("/api/odoo/contact-requests", isAuthenticated, async (req: any, res) => {
  try {
    const { empresaId, userId } = await getUserContext(req);
    if (!empresaId || !userId) {
      return res.status(400).json({
        success: false,
        message: "Utilizador sem empresa ou utilizador associado.",
      });
    }

    const { contactoId, entidadeId, tipo, mensagem } = req.body;

    const created = await storage.createOdooContactRequest({
      empresaId,
      userId,
      contactoId: contactoId || null,
      entidadeId: entidadeId || null,
      tipo,
      mensagem: mensagem || null,
    });

    return res.json({ success: true, data: created });
  } catch (error: any) {
    console.error("[OdooContactRequests] Erro ao criar pedido:", error);
    return res.status(500).json({
      success: false,
      message: error?.message ?? "Erro ao criar pedido de contacto Odoo.",
    });
  }
});

PROPOSITO:
  Criar novo pedido de acesso a contacto/entidade Odoo

AUTENTICACAO:
  isAuthenticated (qualquer user autenticado)

PARAMETROS (body):
  - contactoId (optional): ID do contacto solicitado
  - entidadeId (optional): ID da entidade solicitada
  - tipo (required): "contacto" ou "entidade"
  - mensagem (optional): texto do user

VALIDACOES:
  ✓ empresaId REQUIRED (do context)
  ✓ userId REQUIRED (do context)
  ✓ contactoId/entidadeId coalesced a null se omitidos
  ✓ mensagem coalesced a null se omitido

RESPONSE (sucesso):

ERROR RESPONSE:

app.get("/api/odoo/contact-requests", requireAdmin, async (req: any, res) => {

app.patch(
  "/api/odoo/contact-requests/:id",
  requireAdmin,
  async (req: any, res) => {
    try {
      const { empresaId } = await getUserContext(req);
      if (!empresaId) {
        return res.status(400).json({
          success: false,
          message: "Administrador sem empresa associada.",
        });
      }

      const requestId = req.params.id;
      const { estado } = req.body; // "pendente" | "em_progresso" | "concluido"

      const updated = await storage.updateOdooContactRequestStatus(
        empresaId,
        requestId,
        estado
      );

      if (!updated) {
        return res.status(404).json({
          success: false,
          message: "Pedido não encontrado para esta empresa.",
        });
      }

      return res.json({ success: true, data: updated });
    } catch (error: any) {
      console.error("[OdooContactRequests] Erro ao atualizar pedido:", error);
      return res.status(500).json({
        success: false,
        message:
          error?.message ?? "Erro ao atualizar pedido de contacto Odoo.",
      });
    }
  }
);

PROPOSITO:
  Atualizar estado de um pedido (admin marca como concluído, em progresso, etc.)

AUTENTICACAO:
  requireAdmin (apenas admin)

PARAMETROS:
  - Path: :id (request ID)
  - Body: { estado: "pendente" | "em_progresso" | "concluido" }

VALIDACOES:
  ✓ Admin apenas (via requireAdmin)
  ✓ empresaId REQUIRED (do context)
  ✓ requestId REQUIRED (path param)
  ✓ estado REQUIRED (body)
  ✓ Multi-tenancy: Admin NÃO consegue atualizar pedidos de outra empresa

RESPONSE (sucesso):
  {
    "success": true,
    "data": {
      "id": "uuid",
      "empresaId": "uuid",
      "userId": "uuid",
      "contactoId": "uuid" | null,
      "entidadeId": "uuid" | null,
      "tipo": "contacto" | "entidade",
      "mensagem": "texto" | null,
      "estado": "em_progresso",  ← ALTERADO
      "createdAt": ISO8601 timestamp
    }
  }

ERROR RESPONSES:
  - 400: "Administrador sem empresa associada"
  - 404: "Pedido não encontrado para esta empresa" (id inválido ou wrong empresa)
  - 500: Erro de BD

PADROES:
  ✓ RESTful PATCH (partial update)
  ✓ Null check on result (updated ?? null em storage)
  ✓ 404 se não encontrado (muito melhor que 200 com null)

ARQUITETURA - FLUXO HTTP
================================================================================

CLIENT FRONTEND:
  1. Agent submits form → POST /api/odoo/contact-requests
     Body: { contactoId: "...", tipo: "contacto", mensagem: "..." }
     Response: { success: true, data: { id, estado: "pendente", ... } }

  2. Admin dashboard queries → GET /api/odoo/contact-requests
     Response: { success: true, data: [ { id, userId, tipo, estado, ... }, ... ] }

  3. Admin clicks "Approve" → PATCH /api/odoo/contact-requests/:id
     Body: { estado: "concluido" }
     Response: { success: true, data: { id, estado: "concluido", ... } }

MIDDLEWARE STACK:
  POST:
    Request → isAuthenticated middleware → extract userId/empresaId
              → body validation (Zod in storage layer)
              → storage.createOdooContactRequest()
              → Response

  GET:
    Request → requireAdmin middleware → check is admin
              → extract empresaId → storage.listOdooContactRequestsByEmpresa()
              → Response

  PATCH:
    Request → requireAdmin middleware → check is admin
              → extract empresaId, request id from path
              → body validation (estado enum)
              → storage.updateOdooContactRequestStatus()
              → Response
