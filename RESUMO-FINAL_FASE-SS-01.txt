```typescript
// FASE SS-01: Search Result Types
export type SearchResult<T = any> = {
  id: string;
  label: string;
  extraInfo?: string | null;
  data?: T;
};

export type EntidadeSearchResult = SearchResult<{ cidade?: string; nif?: string }>;
export type ContactoSearchResult = SearchResult<{ entidadeNome?: string; email?: string }>;
export type VisitaSearchResult = SearchResult<{ entidadeNome?: string; date?: string }>;
```

Uso:
- Importar em frontend: `import type { EntidadeSearchResult } from "@shared/schema"`
- Tip: Tipos são exportados em shared/schema.ts para reutilização

ENDPOINTS IMPLEMENTADOS
================================================================================

Ficheiro: server/routes.ts (linhas 3944-4035)

1. GET /api/crm/entidades/search?q=...
   Params:
     - q: string (pesquisa em nome da entidade)
   
   Lógica:
     - Obter todas entidades (respeitando RBAC)
     - Filtrar por q (ILIKE '%q%')
     - Ordenar por nome A-Z
     - Limitar a 20 resultados
   
   Response:
   ```json
   [
     {
       "id": "uuid",
       "label": "Acme Corporation",
       "extraInfo": "Lisboa, PT",
       "data": {
         "cidade": "Lisboa",
         "nif": "PT123456789"
       }
     }
   ]
   ```

2. GET /api/crm/contactos/search?q=...&entidadeId=...
   Params:
     - q: string (pesquisa em nome do contacto)
     - entidadeId: UUID (opcional, filtrar por entidade)
   
   Lógica:
     - Obter contactos (respeitando RBAC + entidadeId)
     - Filtrar por q
     - Ordenar por nome A-Z
     - Limitar a 20 resultados
   
   Response:
   ```json
   [
     {
       "id": "uuid",
       "label": "João Silva",
       "extraInfo": "Acme Corporation",
       "data": {
         "entidadeNome": "Acme Corporation",
         "email": "joao@acme.pt"
       }
     }
   ]
   ```

3. GET /api/crm/visitas/search?q=...&entidadeId=...
   Params:
     - q: string (pesquisa em notas/entidade)
     - entidadeId: UUID (opcional, filtrar por entidade)
   
   Lógica:
     - Obter visitas (respeitando RBAC + entidadeId)
     - Filtrar por q em (notas OR entidade.nome)
     - Ordenar por data (mais recentes primeiro)
     - Limitar a 20 resultados
   
   Response:
   ```json
   [
     {
       "id": "uuid",
       "label": "2025-11-28 – Loja XPTO",
       "extraInfo": "Loja XPTO",
       "data": {
         "entidadeNome": "Loja XPTO",
         "date": "2025-11-28T10:30:00Z"
       }
     }
   ]
   ```

================================================================================
INTEGRAÇÃO NO FRONTEND
================================================================================

Exemplo de uso (React):

```typescript
import { useQuery } from "@tanstack/react-query";
import type { EntidadeSearchResult } from "@shared/schema";

export function SearchEntidades({ q }: { q: string }) {
  const { data } = useQuery({
    queryKey: ["/api/crm/entidades/search", q],
    queryFn: async () => {
      const resp = await fetch(`/api/crm/entidades/search?q=${encodeURIComponent(q)}`, {
        credentials: "include"
      });
      return resp.json() as Promise<EntidadeSearchResult[]>;
    },
    enabled: q.length > 0
  });

  return (
    <ul>
      {data?.map(ent => (
        <li key={ent.id}>
          {ent.label} ({ent.extraInfo})
        </li>
      ))}
    </ul>
  );
}
```

PERFORMANCE E LIMITAÇÕES
================================================================================

✅ Performance:
   - Pesquisa em memória (não DB query)
   - Máx 20 resultados = response pequena
   - Sem join complexo = fast

⚠ Limitações (por design):
   - Search param 'q' é case-insensitive
   - Não suporta regex
   - Sem fuzzy matching
   - Sem score/relevância
   → Adequado para UIs de seleção/autocomplete
