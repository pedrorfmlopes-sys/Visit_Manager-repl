4) (Preparação) Helper para validar Leads – sem usar ainda

Opcional mas recomendado já:

ou

server/integrations/odooClient.ts / server/services/crmService.ts (se quiseres já deixar pronto)

Adicionar algo assim (sem ainda chamar em lado nenhum):

import { db } from "../../shared/db";
import { empresas } from "../../shared/schema";
import { eq } from "drizzle-orm";

export async function assertLeadsEnabled(empresaId: string): Promise<void> {
  const empresa = await db.query.empresas.findFirst({
    where: eq(empresas.id, empresaId),
    columns: {
      id: true,
      crmLeadsEnabled: true,
    },
  });

  if (!empresa?.crmLeadsEnabled) {
    const error: any = new Error("LEADS_NOT_ENABLED");
    error.code = "LEADS_NOT_ENABLED";
    throw error;
  }
}
