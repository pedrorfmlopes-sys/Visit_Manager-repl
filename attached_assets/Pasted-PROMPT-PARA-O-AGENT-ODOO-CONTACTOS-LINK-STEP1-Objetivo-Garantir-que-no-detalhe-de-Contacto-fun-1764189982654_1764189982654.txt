app.post(
  "/api/contactos/:id/odoo-link",
  requireAuth,
  async (req: any, res) => {
    try {
      const { userId, userRole, empresaId } = await getUserContext(req);
      const contactoId = req.params.id;
      const { odooPartnerId } = req.body as { odooPartnerId: number };

      if (!empresaId) {
        return res.status(400).json({ success: false, message: "Empresa em falta" });
      }

      if (!odooPartnerId) {
        return res.status(400).json({
          success: false,
          message: "odooPartnerId em falta",
        });
      }

      // Opcional: validar se o contacto pertence à empresa
      const contacto = await db.query.contactos.findFirst({
        where: and(
          eq(contactos.id, contactoId),
          eq(contactos.empresaId, empresaId)
        ),
      });

      if (!contacto) {
        return res.status(404).json({
          success: false,
          message: "Contacto não encontrado",
        });
      }

      await db
        .update(contactos)
        .set({ odooPartnerId: String(odooPartnerId) })
        .where(
          and(
            eq(contactos.id, contactoId),
            eq(contactos.empresaId, empresaId)
          )
        );

      return res.json({ success: true });
    } catch (error: any) {
      console.error("[Odoo] /contactos/:id/odoo-link error:", {
        message: error?.message,
        stack: error?.stack,
      });

      return res.status(500).json({
        success: false,
        error: "Odoo link contacto error",
        message: error?.message ?? "Erro ao ligar o contacto ao parceiro Odoo.",
      });
    }
  }
);

Função para ligar contacto ao parceiro:

const handleLinkOdooPartnerToContacto = async (partnerId: number) => {
  if (!contacto?.id) return;
  setOdooLinkError(null);

  try {
    const response = await fetch(
      `/api/contactos/${contacto.id}/odoo-link`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({ odooPartnerId: partnerId }),
      }
    );

    const data = await response.json();

    if (!response.ok || data.success === false) {
      throw new Error(data.message || `HTTP ${response.status}`);
    }

    // Atualizar contacto localmente
    if (contacto) {
      contacto.odooPartnerId = String(partnerId);
    }

    // Se houver React Query, invalidar query do contacto
    // queryClient.invalidateQueries({ queryKey: ["/api/contactos", contactoId] });

    toast?.({
      title: "Contacto ligado ao Odoo",
      description: `O contacto foi ligado ao parceiro Odoo #${partnerId}.`,
    });
  } catch (error: any) {
    console.error("[Odoo] Error linking contacto to partner:", error);
    setOdooLinkError(
      error?.message || "Erro ao ligar o contacto ao parceiro Odoo."
    );
  }
};


Função para desligar (se existir botão “Desligar do Odoo”):

const handleUnlinkOdooPartnerFromContacto = async () => {
  if (!contacto?.id) return;
  setOdooLinkError(null);

  try {
    const response = await fetch(
      `/api/contactos/${contacto.id}/odoo-unlink`,
      {
        method: "POST",
        credentials: "include",
      }
    );

    const data = await response.json();

    if (!response.ok || data.success === false) {
      throw new Error(data.message || `HTTP ${response.status}`);
    }

    if (contacto) {
      contacto.odooPartnerId = null;
    }

    toast?.({
      title: "Contacto desligado do Odoo",
      description: "O contacto foi desligado do parceiro Odoo.",
    });
  } catch (error: any) {
    console.error("[Odoo] Error unlinking contacto from partner:", error);
    setOdooLinkError(
      error?.message || "Erro ao desligar o contacto do parceiro Odoo."
    );
  }
};
