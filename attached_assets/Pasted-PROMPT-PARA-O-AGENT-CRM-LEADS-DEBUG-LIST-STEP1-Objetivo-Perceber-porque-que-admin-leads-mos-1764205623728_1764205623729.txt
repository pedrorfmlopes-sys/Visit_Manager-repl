2) Debug da rota GET /api/crm/leads

Ficheiro: server/routes/crmLeads.ts

Na rota GET /api/crm/leads (a lista), adiciona logs antes de devolver a resposta, sem mudar a lógica de negócio:

router.get("/", requireAuth, async (req, res) => {
  try {
    const { empresaId } = await getUserContext(req);
    await assertLeadsEnabled(empresaId);

    const entidadeId =
      typeof req.query.entidadeId === "string" ? req.query.entidadeId : undefined;
    const contactoId =
      typeof req.query.contactoId === "string" ? req.query.contactoId : undefined;
    const visitaId =
      typeof req.query.visitaId === "string" ? req.query.visitaId : undefined;

    let whereClause: any = eq(leads.empresaId, empresaId);

    if (entidadeId) {
      whereClause = and(whereClause, eq(leads.entidadeId, entidadeId));
    }
    if (contactoId) {
      whereClause = and(whereClause, eq(leads.contactoId, contactoId));
    }
    if (visitaId) {
      whereClause = and(whereClause, eq(leads.visitaId, visitaId));
    }

    const rows = await db.query.leads.findMany({
      where: whereClause,
      orderBy: (l, { desc }) => desc(l.createdAt),
    });

    console.log("[CRM Leads] GET /api/crm/leads", {
      empresaId,
      entidadeId,
      contactoId,
      visitaId,
      count: rows.length,
    });

    return res.json({ leads: rows });
  } catch (error: any) {
    ...
  }
});


Corre:

npm run dev

ou leads.empresaId está a null/errado,
