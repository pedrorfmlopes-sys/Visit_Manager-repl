   ✓ setIsUploading(true)

   ✓ formData.append("file", file)

4. POST REQUEST
   ✓ fetch(`/api/crm/leads/${leadId}/odoo/attachments`, {

     - toast SUCCESS

     - fileInputRef.current.value = ""

6. FINALLY
   ✓ setIsUploading(false)

7. UI STATE
   ✓ isUploading = true: input disabled + "A enviar ficheiro..." msg
   ✓ isUploading = false: input enabled
