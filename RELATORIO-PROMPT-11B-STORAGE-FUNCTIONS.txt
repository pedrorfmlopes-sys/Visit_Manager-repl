async listOdooContactRequestsByEmpresa(
  empresaId: string
): Promise<OdooContactRequest[]> {
  return db
    .select()
    .from(odooContactRequests)
    .where(eq(odooContactRequests.empresaId, empresaId))
    .orderBy(desc(odooContactRequests.createdAt));
}

PROPOSITO:
  Listar todos os pedidos de uma empresa (para dashboard de admin)

PARAMETROS:
  - empresaId: string
    ID da empresa para filtrar requests

LOGICA:
  1. SELECT * from odooContactRequests
  2. WHERE empresaId = empresaId (param)
  3. ORDER BY createdAt DESC (mais recentes primeiro)
  4. Retorna array de pedidos

FILTERING:
  ✓ eq(odooContactRequests.empresaId, empresaId)
  ✓ Garante multi-tenancy (cada empresa vê seus próprios pedidos)

ORDERING:
  ✓ desc(odooContactRequests.createdAt)
  ✓ Mais recentes no topo (mais relevant para admin)

RETURN TYPE:
  Promise<OdooContactRequest[]>
  Array de pedidos (pode estar vazio)

PADROES SEGUIDOS:
  ✓ Mesmo estilo de queries (eq, desc usados em contactos, entidades, etc.)
  ✓ Async function
  ✓ Simples, legível, performante

FUNCAO 3: updateOdooContactRequestStatus (Linhas 1854-1871)
================================================================================

async updateOdooContactRequestStatus(
  empresaId: string,
  requestId: string,
  estado: "pendente" | "em_progresso" | "concluido"
): Promise<OdooContactRequest | null> {
  const [updated] = await db
    .update(odooContactRequests)
    .set({ estado })
    .where(
      and(
        eq(odooContactRequests.id, requestId),
        eq(odooContactRequests.empresaId, empresaId)
      )
    )
    .returning();

  return updated ?? null;
}

PROPOSITO:
  Atualizar estado de um pedido (admin marca como concluido, etc.)

PARAMETROS:
  - empresaId: string
    ID da empresa (validacao de propriedade)
  - requestId: string
    ID do pedido a atualizar
  - estado: "pendente" | "em_progresso" | "concluido"
    Novo estado

LOGICA:
  1. UPDATE odooContactRequests
  2. SET estado = novo estado
  3. WHERE id = requestId AND empresaId = empresaId
  4. Retorna record atualizado OU null se não encontrado

WHERE CLAUSE:
  ✓ and(eq(...), eq(...))
  ✓ Dois criterios:
    - id do pedido (exatidao)
    - empresaId (segurança multi-tenancy)
  ✓ Admin não consegue atualizar pedidos de outra empresa

RETURNING:
  ✓ Usa .returning() para obter record atualizado
  ✓ Destructura [updated] do array
  ✓ Retorna updated ?? null (tratamento de not found)

RETURN TYPE:
  Promise<OdooContactRequest | null>
  - OdooContactRequest: record foi atualizado
  - null: pedido não encontrado (id inválido ou não pertence a empresa)

PADROES SEGUIDOS:
  ✓ Mesmo estilo de UPDATE (como updateEntidade, updateContacto, etc.)
  ✓ Validacao de proprietario com and()
  ✓ Null-safe return
  ✓ Type-safe estado

EXEMPLO USAGE:
  const updated = await storage.updateOdooContactRequestStatus(
    "empresa-123",
    "request-456",
    "concluido"
  );
  // Se success: updated.estado = "concluido"
  // Se not found: updated = null
