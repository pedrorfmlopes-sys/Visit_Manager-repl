DEPOIS:

  export function useAuth() {
    const { data: user, isLoading } = useQuery<AuthUser>({
      queryKey: ["/api/auth/user"],
      retry: false,
    });

    const isAdmin = user?.role === "admin";
    const empresa = user?.empresa;

    // PROMPT 7: Calcular flags derivadas para Odoo Contacts
    const odooCrmEnabled = empresa?.odooCrmEnabled ?? false;
    const odooContactsFeatureEnabled = empresa?.odooContactsFeatureEnabled ?? false;
    const odooContactsAdminEnabled = empresa?.odooContactsAdminEnabled ?? true;
    const odooContactsAgentsEnabled = empresa?.odooContactsAgentsEnabled ?? false;
    const crmVisitsOdooSyncEnabled = empresa?.crmVisitsOdooSyncEnabled ?? false;

    // PROMPT 7: Derivadas - Permissões RBAC (3-tier)
    const canUseOdooContacts =
      odooCrmEnabled &&
      odooContactsFeatureEnabled &&
      (isAdmin ? odooContactsAdminEnabled : odooContactsAgentsEnabled);

    const canSyncVisitsWithOdoo =
      odooCrmEnabled &&
      odooContactsFeatureEnabled &&
      crmVisitsOdooSyncEnabled &&
      (isAdmin ? odooContactsAdminEnabled : odooContactsAgentsEnabled);

    return {
      user,
      isLoading,
      isAuthenticated: !!user,
      isAdmin,
      empresa,
      // PROMPT 7: Odoo Contacts derived permissions
      canUseOdooContacts,
      canSyncVisitsWithOdoo,
    };
  }

LOGICA - 3-TIER RBAC IMPLEMENTATION
================================================================================

TIER 1 - ODOO CRM GLOBAL:
  Condicao: odooCrmEnabled === true
  Efeito: Desativa tudo se false
  Controlo: Admin em AdminEmpresa

TIER 2 - FEATURE GATE:
  Condicao: odooContactsFeatureEnabled === true
  Efeito: Desativa features se false
  Controlo: Admin em AdminEmpresa

TIER 3 - RBAC ROLE-BASED:
  Admin:
    Condicao: odooContactsAdminEnabled === true
    Efeito: Admin pode usar se true
    Controlo: Admin em AdminEmpresa
  
  Agent:
    Condicao: odooContactsAgentsEnabled === true
    Efeito: Agent pode usar se true
    Controlo: Admin em AdminEmpresa

TIER 4 - OPERATION SPECIFIC (Sync):
  Condicao: crmVisitsOdooSyncEnabled === true
  Efeito: Sync apenas se true
  Controlo: Admin em AdminEmpresa

FLUXO PERMISSOES:

  canUseOdooContacts = true se:
    ✓ Tier 1 (CRM) = ON
    ✓ Tier 2 (Feature) = ON
    ✓ Tier 3 (RBAC) = ON (admin ? adminEnabled : agentsEnabled)

  canSyncVisitsWithOdoo = true se:
    ✓ Tier 1 (CRM) = ON
    ✓ Tier 2 (Feature) = ON
    ✓ Tier 4 (Sync) = ON
    ✓ Tier 3 (RBAC) = ON (admin ? adminEnabled : agentsEnabled)
