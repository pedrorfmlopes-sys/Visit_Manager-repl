  export async function assertLeadsEnabled(empresaId?: string | null): Promise<void> {
    if (!empresaId) {
      // Sem empresa ID → skip check (graceful)
      return;
    }

    try {
      // Fetch empresa da database
      const empresa = await empresasStorage.getById(empresaId);
      
      if (!empresa) {
        const error = new Error("EMPRESA_NOT_FOUND");
        (error as any).code = "EMPRESA_NOT_FOUND";
        throw error;
      }

      // Check flag (fonte da verdade)
      const isEnabled = (empresa as any).crmLeadsEnabled ?? false;
      
      if (!isEnabled) {
        const error = new Error("CRM_LEADS_DISABLED");
        (error as any).code = "CRM_LEADS_DISABLED";
        throw error;
      }
      
      // Success - modulo esta ativado
    } catch (error) {
      throw error; // Re-throw para handler tratar
    }
  }

COMPORTAMENTO:
  Input: empresaId = "xxx"
  ├─ Se empresa.crmLeadsEnabled = true
  │  → Returns (sem erro)
  │  → Handler continua
  ├─ Se empresa.crmLeadsEnabled = false
  │  → Throws { code: "CRM_LEADS_DISABLED" }
  │  → Handler salta para catch
  └─ Se empresa nao existe
     → Throws { code: "EMPRESA_NOT_FOUND" }
