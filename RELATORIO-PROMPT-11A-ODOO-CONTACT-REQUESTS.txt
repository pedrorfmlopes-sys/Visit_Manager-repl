    - OnDelete: CASCADE (se empresa deleted, request deletado)
    - Razao: Empresa MUST exist

  userId:
    - Type: varchar
    - Constraint: NOT NULL, REFERENCES users.id
    - OnDelete: CASCADE (se user deleted, request deletado)
    - Razao: User MUST exist

  contactoId:
    - Type: varchar
    - Constraint: NULLABLE, REFERENCES contactos.id
    - OnDelete: SET NULL (se contacto deleted, campo nullificado)
    - Razao: Request pode ser para contacto OU entidade (not both)

  entidadeId:
    - Type: varchar
    - Constraint: NULLABLE, REFERENCES entidades.id
    - OnDelete: SET NULL (se entidade deleted, campo nullificado)
    - Razao: Request pode ser para entidade OU contacto (not both)

  Tipo: text
  Values: "contacto" | "entidade"
  Constraint: NOT NULL
  Razao: Identifica qual tipo de acesso está sendo requestado

  Constraint: NULLABLE
  Razao: User pode deixar mensagem vazia ou incluir contexto/justificacao
