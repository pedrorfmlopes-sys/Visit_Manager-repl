    try {
      setSavingLeads(true);

        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        credentials: "include",
        body: JSON.stringify({
          crmLeadsEnabled: leadsEnabled,
        }),
      });

      if (!resp.ok) {
        throw new Error(`HTTP ${resp.status}`);
      }

      toast({
        title: "Definições de Leads guardadas",
        description: leadsEnabled
          ? "Módulo de Leads está ativo para esta empresa."
          : "Módulo de Leads foi desativado.",
      });

      // Refresh data after successful save
      queryClient.invalidateQueries({ queryKey: ["/api/admin/empresa"] });
      queryClient.invalidateQueries({ queryKey: ["/api/auth/user"] });
    } catch (error: any) {
      console.error("[Settings] Error saving CRM leads flag:", error);
      toast({
        title: "Erro ao guardar definições de Leads",
        description: "Verifica a ligação e tenta novamente.",
        variant: "destructive",
      });
    } finally {
      setSavingLeads(false);
    }
  };

================================================================================
IMPLEMENTACAO TECHNICAL
================================================================================

PADRÃO USADO:

  1. Import hook:
     import { useQueryClient } from "@tanstack/react-query";

  2. Call hook no top-level:
     const queryClient = useQueryClient();

  3. Use em handler/callback:
     queryClient.invalidateQueries({ queryKey: [...] });

WHY THIS WORKS:

  React Query v5 internals:
    - QueryClient é singleton (created at app root)
    - useQueryClient() é hook que dá acesso via context
    - invalidateQueries() marca todas queries com key como stale
    - Auto-refetch acontece se query é ativa (mounted)
    - Re-render acontece quando dados mudam

GARANTIAS:

  ✅ Synchronous: Invalidation é immediate
  ✅ Reactive: Listeners notificados em tempo real
  ✅ Type-safe: useQueryClient() retorna QueryClient typed
  ✅ Idiomatic: Segue React Hooks patterns
  ✅ Best practice: Recomendado por React Query docs
